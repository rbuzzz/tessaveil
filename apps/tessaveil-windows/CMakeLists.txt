cmake_minimum_required(VERSION 3.31)
project(TessaveilWindowsAlpha LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets Test)
get_target_property(qt_kind Qt6::Core TYPE)
if(NOT qt_kind STREQUAL "STATIC_LIBRARY")
 message(FATAL_ERROR "Static Qt required; binary redistribution is NOT approved")
endif()
if(NOT DEFINED CONTROLLER_LIBRARY)
 message(FATAL_ERROR "Pass the external release Rust static library")
endif()
add_library(controller STATIC IMPORTED)
set_target_properties(controller PROPERTIES IMPORTED_LOCATION "${CONTROLLER_LIBRARY}")
add_library(alpha_ui STATIC window.cpp)
target_link_libraries(alpha_ui PUBLIC Qt6::Widgets controller bcrypt ntdll userenv ws2_32 dbghelp)
qt_add_executable(Tessaveil WIN32 main.cpp)
target_link_libraries(Tessaveil PRIVATE alpha_ui wtsapi32)
if(DEFINED ALPHA_LINK_MAP)
 target_link_options(Tessaveil PRIVATE "-Wl,-Map,${ALPHA_LINK_MAP}")
endif()
qt_add_executable(alpha_ui_tests tests/ui.cpp)
target_link_libraries(alpha_ui_tests PRIVATE alpha_ui Qt6::Test wtsapi32)
foreach(target Tessaveil alpha_ui_tests)
 target_link_options(${target} PRIVATE -static)
 qt_import_plugins(${target} INCLUDE Qt6::QWindowsIntegrationPlugin)
endforeach()
enable_testing()
add_test(NAME native_workflow COMMAND alpha_ui_tests)
