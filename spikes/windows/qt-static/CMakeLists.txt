cmake_minimum_required(VERSION 3.31)
project(TessaveilQtProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets)
get_target_property(qt_kind Qt6::Core TYPE)
if(NOT qt_kind STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "Probe requires static Qt; dynamic Qt is not equivalent evidence")
endif()
add_library(probe_core STATIC probe_core.cpp)
qt_add_executable(TessaveilQtProbe WIN32 main.cpp)
target_link_libraries(TessaveilQtProbe PRIVATE Qt6::Widgets probe_core)
if(MINGW)
  target_link_options(TessaveilQtProbe PRIVATE -static)
endif()
qt_import_plugins(TessaveilQtProbe INCLUDE Qt6::QWindowsIntegrationPlugin)
